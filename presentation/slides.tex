    \end{itemize}

    \hl{148} features \quad \hl{50/50} classes

\begin{frame}{Our decision tree \owner{B}}
  \begin{columns}[T]
    \column{0.52\textwidth}
    \begin{itemize}
      \item Criteria: \textbf{Gini}, \textbf{entropy} (Task B), \textbf{MSE} (Task A)
      \item Split = max impurity decrease\\
        $\Delta I = I(P)-\frac{N_L}{N}I(L)-\frac{N_R}{N}I(R)$
      \item \textbf{Vectorised search}: sort once, cumulative counts / sums
        $\Rightarrow$ all thresholds scored at once
      \item 4 controls: \texttt{max\_depth}, \texttt{min\_samples\_split},
        \texttt{min\_samples\_leaf}, \texttt{min\_impurity\_decrease}
      \item \texttt{predict\_proba}, \texttt{feature\_importances\_}
    \end{itemize}
    \column{0.45\textwidth}
    \includegraphics[width=\linewidth]{tree_B_depth.png}\\
    {\scriptsize\color{muted}Train (dashed) vs.\ validation F1: deeper $\Rightarrow$ overfits, validation flattens after depth 12.}
  \end{columns}
\end{frame}

% 5 ----------------------------------------------------------------------
\begin{frame}{Our Pegasos SVM \owner{C}}
  \begin{columns}[T]
    \column{0.52\textwidth}
    \small
    $J(\mathbf w,b)=\frac{\lambda}{2}\lVert\mathbf w\rVert^2+\frac1n\sum_i\max(0,1-y_i(\mathbf w^\top\mathbf x_i+b))$
    \begin{itemize}
      \item Mini-batch sub-gradient; hinge-active points push $\mathbf w$
      \item \alert{Fix}: paper's $\eta_t=1/(\lambda t)$ throws the bias to $\pm$hundreds
        $\Rightarrow$ $\eta_t=1/(\lambda(t+1/\lambda))$
      \item Reports $\lVert\mathbf w\rVert$, margin $2/\lVert\mathbf w\rVert$, support vectors
      \item \textbf{Non-linear}: random Fourier features\\
        $\phi(\mathbf x)\cdot\phi(\mathbf x')\approx e^{-\gamma\lVert\mathbf x-\mathbf x'\rVert^2}$,
        cost $O(nD)$ not $O(n^2)$
    \end{itemize}
    \column{0.45\textwidth}
    \includegraphics[width=\linewidth]{svm_objective.png}\\
    {\scriptsize\color{muted}Objective decreases; the paper schedule (orange) spikes first.}
  \end{columns}
\end{frame}

% 6 ----------------------------------------------------------------------
\begin{frame}{Experimental setup}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{One protocol for every model}
    \begin{itemize}
      \item Same stratified split 70/15/15 (seed 42)\\
            40,605 / 8,701 / 8,701
      \item Threshold and scaling from \textbf{train only}
      \item Hyperparameters chosen on \textbf{validation}
      \item Test scored \textbf{once}
    \end{itemize}
    \column{0.47\textwidth}
    \textbf{Fair baselines}
    \begin{itemize}
      \item sklearn trees: same depth / leaf / criterion
      \item \texttt{LinearSVC}, SGD: $C=1/(\lambda n)$ $\Rightarrow$ same objective
      \item RBF \texttt{SVC} (10k subsample)
      \item 5-fold CV mean $\pm$ std, fit/predict time
    \end{itemize}
  \end{columns}
  \vspace{1em}
  \textbf{Metrics} (own NumPy code): RMSE, MAE, $R^2$ \,|\, precision, recall, F1, ROC-AUC\\
  \textbf{Reproducible}: \texttt{python -m src.run\_all} rebuilds everything in $\sim$24 min; 49 unit tests
\end{frame}

\begin{frame}{What we learned}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \begin{itemize}
      \item \textbf{Tree wins F1}: price tier = area $\times$ district interaction, axis splits capture it
      \item \textbf{SVM wins AUC}: continuous margin ranks listings smoothly
      \item \textbf{Margin trade-off}: larger $\lambda$ $\Rightarrow$ wider margin, more support vectors; best at $\lambda=10^{-3}$
      \item \alert{Scaling matters}: standardised rare-district dummies ($|x|\approx200$) break RBF;
        RFF val F1 \textbf{.859} $\to$ \textbf{.913} with 0/1 dummies
    \end{itemize}
    \column{0.47\textwidth}
    \includegraphics[width=\linewidth]{svm_lambda_sweep.png}\\
    {\scriptsize\color{muted}Linear Pegasos: validation score and margin vs.\ $\lambda$.}
  \end{columns}
\end{frame}
